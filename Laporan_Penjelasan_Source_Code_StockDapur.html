function setTheme(tema) {

async function simpanBahan(e) {
  e.preventDefault();
  const payload = {
    namaBahan:   el('add-nama').value.trim(),
    satuan:      el('add-satuan').value.trim(),
    stok:        Number(el('add-stok').value) || 0,
    minStok:     Number(el('add-min').value) || 0,
    hargaSatuan: parseInt(el('add-harga').value) || 0
  };

  if (idSedangEdit !== null) {
    const hasil = await ubahBahanApi(idSedangEdit, payload);
    if (hasil.ok) {
      const idx = daftarBahan.findIndex(b => b._id === idSedangEdit);
      if (idx !== -1) daftarBahan[idx] = hasil.bahan;
      toast('Bahan berhasil diubah');
    }
  } else {
    const hasil = await tambahBahanApi(payload);
    if (hasil.ok) { daftarBahan.push(hasil.bahan); toast('Bahan berhasil ditambahkan'); }
  }
  resetFormBahan();
  el('details-add').open = false;
  render();
}
Penjelasan Source Code: Fungsi percabangan logika yang membedakan apakah pengguna sedang mendaftarkan bahan baru atau memperbarui data bahan yang sudah tersimpan di database.
  • e.preventDefault() Mencegah form melakukan reload halaman web secara default.
  • if (idSedangEdit !== null) Jika variabel penampung ID sedang terisi, sistem mengeksekusi API pembaruan (PUT) dan mengupdate elemen pada state array secara tepat sasaran.
  • daftarBahan.push(hasil.bahan) Jika penambahan baru, dokumen bahan baru yang baru dibuat di MongoDB langsung disisipkan ke state tanpa reload.

async function prosesCatat(e) {
  e.preventDefault();
  const id      = el('catat-bahan').value;
  const jumlah  = Number(el('catat-jumlah').value);
  const catatan = el('catat-catatan').value.trim();
  const errEl   = el('catat-error');

  if (!id || jumlah <= 0) {
    errEl.textContent = 'Masukkan jumlah yang lebih besar dari 0.';
    errEl.style.display = 'block';
    return;
  }

  const hasil = await catatApi(id, { tipe: tipeCatatat, jumlah, catatan });
  if (!hasil.ok) {
    errEl.textContent = hasil.pesan;
    errEl.style.display = 'block';
    return;
  }

  errEl.style.display = 'none';
  const idx = daftarBahan.findIndex(b => b._id === id);
  if (idx !== -1) daftarBahan[idx] = hasil.bahan;
  riwayat.push(hasil.transaksi);

  const verb = tipeCatatat === 'masuk' ? 'ditambahkan' : 'dikeluarkan';
  toast(`${formatJumlah(jumlah)} ${hasil.bahan.satuan} ${hasil.bahan.nama_bahan} berhasil ${verb}`);
  el('form-catat').reset();
  render();
}
Penjelasan Source Code: Inti pengendali mutasi barang keluar-masuk yang bertugas memvalidasi nilai input, mengirimkan transaksi ke server, dan memperbarui riwayat jurnal.
  • if (!id || jumlah <= 0) Validasi tingkat awal (*client-side validation*) agar peramban tidak membuang bandwidth dengan request bernilai tidak valid.
  • await catatApi(id, { tipe: tipeCatatat, jumlah, catatan }) Menghubungi API backend untuk memodifikasi saldo stok bahan secara aman di database.
  • riwayat.push(hasil.transaksi) Menambahkan bukti transaksi baru ke array riwayat lokal agar dapat langsung dipantau di tab Riwayat Terakhir dan tab Jurnal.

export function renderBon(bahanList) {
  const menipis = bahanList.filter(b => b.stok < b.min_stok);
  let html = `<div class="bon-title">Bon belanja</div><div class="bon-date">${hariIni()}</div>`;

  if (!menipis.length) {
    html += `<hr class="bon-divider"><div class="bon-empty">Semua stok aman 👍</div>`;
  } else {
    let total = 0;
    html += `<hr class="bon-divider">`;
    menipis.forEach(b => {
      const perlu = b.min_stok - b.stok;
      const biaya = perlu * b.harga_satuan;
      total += biaya;
      html += `<div class="bon-line"><span class="bon-name">${b.nama_bahan}</span><span class="bon-detail">${formatJumlah(perlu)} ${b.satuan} <span class="bon-price">${formatRp(biaya)}</span></span></div>`;
    });
    html += `<hr class="bon-divider"><div class="bon-total"><span>Total</span><span class="bon-price">${formatRp(total)}</span></div>`;
    html += `<div class="bon-note">Estimasi beli sampai batas minimum.</div>`;
  }
  document.getElementById('bon-belanja').innerHTML = html;
}
Penjelasan Source Code: Fungsi cerdas yang menyaring bahan-bahan kritis dan otomatis menghitung kekurangan stok serta estimasi uang belanja yang harus disiapkan.
  • bahanList.filter(b => b.stok < b.min_stok) Menyaring bahan baku yang berada di bawah stok minimum.
  • const perlu = b.min_stok - b.stok; Menghitung jumlah defisit kuantitas yang perlu dibeli ke pasar untuk memenuhi kembali batas minimum.
  • const biaya = perlu * b.harga_satuan; Mengalikan kebutuhan belanja dengan harga per unit untuk mendapatkan total estimasi anggaran.

Contoh 14: Bagian Backend Node.js (server.js - Middleware Penjaga Route harusLogin)

(Di sini kamu tempel screenshot potongan kodingan harusLogin)
javascript code snippet 
async function harusLogin(req, res, next) {
  const auth = req.headers.authorization;
  if (!auth || !auth.startsWith('Basic ')) {
    return res.status(401).json({ ok: false, pesan: 'Belum login.' });
  }
  try {
    const decoded  = Buffer.from(auth.slice(6), 'base64').toString();
    const idx      = decoded.indexOf(':');
    const username = decoded.substring(0, idx);
    const password = decoded.substring(idx + 1);
    const db       = await getDB();
    const user     = await db.collection('users').findOne({ username, password });
    if (!user) return res.status(401).json({ ok: false, pesan: 'Belum login.' });
    req.pengguna = { id: user._id, username: user.username, nama: user.nama, role: user.role };
    next();
  } catch (e) {
    return res.status(401).json({ ok: false, pesan: 'Belum login.' });
  }
}
Penjelasan Source Code: Middleware keamanan (*security gatekeeper*) yang memverifikasi token kredensial Basic Auth pada setiap rute privat sebelum operasi database diizinkan.
  • Buffer.from(auth.slice(6), 'base64').toString() Mendekode muatan string Base64 menjadi format username dan password asli.
  • await db.collection('users').findOne(...) Memastikan akun staf benar-benar ada di database sebelum request diproses.
  • req.pengguna = { ... }; next(); Menyematkan identitas staf ke objek request dan meneruskan eksekusi ke pengendali rute berikutnya.

app.post('/api/stok/:id/catat', harusLogin, async (req, res) => {
  const db = await getDB();
  const { tipe, jumlah, catatan } = req.body;
  const bahan = await db.collection('bahan').findOne({ _id: new ObjectId(req.params.id) });
  if (!bahan) return res.status(404).json({ ok: false, pesan: 'Bahan tidak ditemukan.' });
  if (tipe === 'keluar' && Number(jumlah) > bahan.stok)
    return res.status(400).json({ ok: false, pesan: 'Stok tidak cukup.' });
  const stokBaru = tipe === 'masuk' ? bahan.stok + Number(jumlah) : bahan.stok - Number(jumlah);
  await db.collection('bahan').updateOne({ _id: new ObjectId(req.params.id) }, { $set: { stok: stokBaru } });
  const bahanTerbaru = await db.collection('bahan').findOne({ _id: new ObjectId(req.params.id) });
  res.json({
    ok: true,
    bahan: bahanTerbaru,
    transaksi: {
      waktu: new Date().toISOString(), tipe,
      namaBahan: bahan.nama_bahan, jumlah: Number(jumlah),
      satuan: bahan.satuan, stokSetelah: stokBaru,
      oleh: req.pengguna.nama, catatan: catatan || ''
    }
  });
});
Penjelasan Source Code: Endpoint krusial yang menjamin konsistensi data stok secara atomik dan mencegah terjadinya sisa stok bernilai negatif.
  • if (tipe === 'keluar' && Number(jumlah) > bahan.stok) Perlindungan integritas data di server; mencegah staf mengeluarkan bahan melebihi stok yang ada di gudang/dapur.
  • db.collection('bahan').updateOne(...) Melakukan update parsial menggunakan operator $set MongoDB untuk merefleksikan saldo kuantitas stok baru.
